```typescript
/**
 * Export the map canvas as an image
 * Called by DashboardCard's export functionality
 */
async function exportMap(config: Partial<ExportConfig> = {}): Promise<ExportResult | null> {
  if (!mapInstance.value) {
    console.warn('Map not initialized for export')
    return null
  }

  try {
    return await exportMapCanvas(mapInstance.value, config)
  } catch (error) {
    console.error('Map export failed:', error)
    return null
  }
}

// Expose to parent via defineExpose (Vue 3) or via emit
```

In DashboardCard's handleExport function, check for map:

```typescript
async function handleExport(format: 'png' | 'svg' = 'png') {
  if (!contentWrapper.value) return

  // Check for Plotly chart first
  const plotElement = contentWrapper.value.querySelector('.js-plotly-plot') as HTMLElement
  if (plotElement) {
    // ... existing Plotly export code ...
    return
  }

  // Check for MapLibre canvas
  const mapContainer = contentWrapper.value.querySelector('[data-exportable-map]')
  const mapCanvas = mapContainer?.querySelector('canvas') as HTMLCanvasElement
  if (mapCanvas) {
    try {
      const filename = props.card.exportName || sanitizeFilename(props.card.title, 'map')
      const dataUrl = mapCanvas.toDataURL('image/png')

      // Trigger download
      const link = document.createElement('a')
      link.href = dataUrl
      link.download = `${filename}.png`
      document.body.appendChild(link)
      link.click()
      document.body.removeChild(link)
    } catch (error) {
      console.error('Map export failed:', error)
    }
    return
  }

  console.warn('No exportable element found in card')
}
```

Note: Map canvas export is simpler than Plotly - we just use canvas.toDataURL() directly. SVG export is not supported for maps (WebGL canvas limitation).
